# Hunter's Eye - user settings
# =============================
#
# Edit this file to change WHEN the program drinks and WHICH key it presses.
# No code changes needed. Save the file and restart the program to apply.
#
# If this file is missing, unreadable, or a value is nonsense, the program falls
# back to its built-in defaults and says so on the console - it will not crash
# and it will not silently run with a setting you did not intend.
#
# See also:
#   assets/targets.txt  - item names to watch for on the ground
#   assets/meters.json  - where the health/mana orbs are (run calibrate_meters.py)


# -----------------------------------------------------------------------------
# [settings] - overall behaviour
# -----------------------------------------------------------------------------
[settings]

# Turn potion drinking off entirely without deleting your rules below.
enabled = yes

# Shortest time between ANY two potion presses, in seconds. This is a SAFETY NET,
# not a tuning knob - it is what stops a mistuned threshold or a misread orb from
# emptying the whole belt in a second. Tune the per-rule "cooldown" instead.
min_gap = 0.6

# How long the F6 hotkey suppresses drinking, in seconds. It re-arms itself
# afterwards on purpose: a snooze you forget about costs you 10 seconds, whereas
# an on/off toggle you forget to switch back on costs you the rest of the session.
snooze = 10

# Readings below this fraction are treated as "cannot read" rather than as a real
# value, and nothing acts on them. A meter does not teleport - health falls
# THROUGH 20%, 15% and 10% on its way to 2%, at about 50 samples a second, so a
# rule watching for "below 20%" has already fired long before the reading gets
# this low. What this filters out is the other thing that produces a very low
# number: a stray patch of the right colour where the meter should be, when the
# meter is not actually on screen (the game lobby, a cutscene, a menu). Set to 0
# to disable and act on any reading at all.
ignore_below = 0.03


# --- Making a new game (F3 / next_game) --------------------------------------

# Put a password on every new game this makes. Off by default, because the safe
# thing for an automatic sequence is to leave a field ALONE unless it is told
# otherwise - and because an unwanted password is invisible until someone cannot
# join your game. With this off, any password already in the box is CLEARED, so
# a password you set by hand earlier does not silently carry into every new game.
use_password = yes

# The password to use when use_password is yes. Taken exactly as written.
password = 123

# Normally the game name is read off the lobby's own "Game Name" box, which
# Diablo II pre-fills with your last game, and then the number on the end is
# incremented ("z25pin35" -> "z25pin36"; a name with no number gets a "0").
# Set this to force a starting name instead - useful on a fresh client where the
# box is empty, or to switch naming scheme without typing it in yourself.
# Leave it blank to read the box, which is what you normally want.
game_name =


# --- Teleporting onto a drop (auto-collect) ----------------------------------

# The key that teleports to wherever the mouse is pointing. When this is set,
# auto-collect points at an item it is about to pick up, presses this key, and
# only then goes for the item - on a public game the drop belongs to whoever
# gets there first, so arriving instantly and picking up a beat later beats
# walking over and picking up on arrival.
#
# Leave it BLANK to switch this off entirely and just walk, which is what the
# program did before this setting existed.
#
# It is never used on an item marked "[key:e]" in assets/targets.txt. Those are
# taken from across the room with telekinesis, so teleporting to them first
# would spend mana removing the exact distance that skill exists to cover.
teleport_key = f

# How far away an item has to be before teleporting to it is worth it, measured
# from your character and given as a fraction of the screen's height ("25%" or
# "0.25"). Anything closer is just clicked and walked to.
#
# A teleport is not free - about half a second of aiming and camera settle, plus
# finding the item again once the whole scene has moved, plus the mana. Inside a
# few feet, walking really is faster, so this is the knob for where "a few feet"
# is. At 1080p, 25% is about 270 pixels, or roughly one and a half character
# heights. Raise it if it teleports onto things it could have stepped on; lower
# it if it walks to things it should have jumped to.
teleport_min_distance = 25%


# --- Grabbing an item with a skill key (auto-collect) ------------------------

# Seconds to wait before trying again when an item tagged "[key:e]" in
# assets/targets.txt does not come to you on the first press.
#
# This is SEPARATE from how often a clicked item is re-clicked (0.8s, in
# main.py), and the two are paced by completely different things. A click means
# "walk over there and pick it up on arrival", so re-clicking faster than the
# walk takes keeps re-issuing the move command and the character never gets
# there. A cast is over the moment it lands, so the only thing pacing a retry is
# how fast your character can cast - which is faster cast rate, which is gear.
#
# Raise this if the potion sometimes comes to you and the extra casts are wasted
# mana; lower it if a failed grab sits there while you wait to try again. It
# cannot usefully go below about 0.25s: each attempt already spends 0.20s
# letting the game notice the cursor before the key is pressed at all
# (AIM_SETTLE_SECONDS in core/actions.py).
key_collect_retry = 0.35


# --- Aiming: how long the cursor sits on a target before acting --------------

# The pause between the cursor arriving on a target and the action firing, in
# seconds. Two values, because the two events are not the same shape:
#
#   click_settle  a click's button-down is itself a MOUSE event and carries the
#                 cursor position in its own Win32 message, behind the move on
#                 the same path and in order - the game cannot process the press
#                 without having the new position in hand.
#   aim_settle    a keypress is a KEYBOARD event and carries no coordinates at
#                 all. The game resolves it against whatever it last decided the
#                 cursor was over, read separately on its own tick, with nothing
#                 tying that read to our move. So it needs longer.
#
# These are here rather than in the code because the right number depends on the
# game and the machine, and the only way to find it is to try values while
# playing. RAISE THEM IF THE FIRST ATTEMPT KEEPS MISSING and the second works -
# the console now says which attempt each pickup landed on, so you can see the
# effect directly rather than guessing. Lower them once it lands first try: the
# cursor is taken away from you for this long on every single action.
click_settle = 0.05
aim_settle = 0.20


# -----------------------------------------------------------------------------
# Potion rules - one [section] each
# -----------------------------------------------------------------------------
# ORDER MATTERS. Rules are checked top to bottom and the FIRST one that can
# actually be acted on wins, so put the most urgent one first.
#
# Each rule takes:
#   meter    - which meter to watch. Must be a name from assets/meters.json
#              (currently: health, mana).
#   below    - drink when the meter is at or below this. Percent ("20%") or a
#              fraction ("0.20") - both mean the same thing.
#   use      - WHICH POTION to drink: healing, mana or rejuvenation. The key is
#              worked out by LOOKING AT YOUR BELT, so you never write a key down
#              and it is always right, even after you rearrange the slots. If
#              two slots hold that potion, the presses alternate between them.
#   keys     - only a fallback, for if the belt cannot be read at all (a bad
#              assets/belt.json, or the game at a resolution it was not
#              calibrated for). Leave it as-is; "use" is what normally decides.
#   cooldown - seconds to wait before THIS rule may fire again. Each rule has
#              its own, which is what lets an emergency potion fire immediately
#              after an ordinary one. Set it longer than the potion takes to
#              work, or the next check still sees a low orb and drinks again.
#   only_if_missing - makes the rule a LAST RESORT: it fires only when your belt
#              has none of the named potion left. See [health_backup] below.
#
# A RULE THAT CANNOT BE ACTED ON IS SKIPPED, NOT OBEYED. If a rule says to drink
# a healing potion and you have none, the program moves on to the next rule
# instead of pressing a dead key - which is what makes the backup rule below
# work, and what stops it pressing "4" for mana when slot 4 is empty.


# Emergency. Rejuvenation fills health AND mana instantly, so it is the right
# thing when health is genuinely low - but it is also the rarest potion, which
# is why nothing else in this file spends one.
[rejuvenation]
meter = health
below = 30%
use = rejuvenation
keys = 2, 3
cooldown = 1.0

# Ordinary healing. Heals over several seconds rather than instantly, so the
# cooldown has to outlast the heal - re-checking before it lands reads a still
# low orb and drinks again, which is how a belt empties in one second.
[health]
meter = health
below = 70%
use = healing
keys = 1
cooldown = 4.0

# BACKUP, for when the healing potions have run out. Waits until 50% rather than
# firing at the 70% above, because a rejuvenation potion is worth more than the
# 20% of health you would be spending it on - and it only exists at all while
# there is no healing potion in the belt, which is what "only_if_missing" says.
[health_backup]
meter = health
below = 50%
use = rejuvenation
only_if_missing = healing
cooldown = 1.0

# Mana. There is deliberately NO backup rule here: if the mana potions run out,
# mana is allowed to hit zero. A rejuvenation potion is far too valuable to
# spend refilling mana, and that is a judgement call, not an oversight - drink
# one by hand if a fight really needs it.
[mana]
meter = mana
below = 20%
use = mana
cooldown = 4.0
